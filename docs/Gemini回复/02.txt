```javascript
      },
      {
        path: 'favorites',
        name: 'Favorites',
        component: () => import('../views/Favorites.vue')
      },
      {
        path: 'search',
        name: 'Search',
        component: () => import('../views/Search.vue')
      },
      {
        path: 'trash',
        name: 'Trash',
        component: () => import('../views/Trash.vue')
      }
    ]
  },
  {
    path: '/share/:code',
    name: 'SharedView',
    component: () => import('../views/SharedView.vue'),
    meta: { requireAuth: false }
  }
];

const router = createRouter({
  history: createWebHistory(),
  routes
});

// 核心路由导航守卫
router.beforeEach((to, from, next) => {
  const userStore = useUserStore();
  const token = userStore.token;

  if (to.meta.requireAuth && !token) {
    // 未登录跳转登录页
    next({ name: 'Login' });
  } else if (token && to.name === 'Login') {
    // 已登录状态下不能访问登录页
    next({ name: 'FileBrowser' });
  } else if (token && userStore.firstLogin && to.name !== 'SetPassword') {
    // 首次登录，强制去设置密码
    next({ name: 'SetPassword' });
  } else {
    next();
  }
});

export default router;
```

#### 1. `src/utils/format.js`

```javascript
export function formatBytes(bytes, decimals = 2) {
  if (bytes === 0) return '0 B';
  const k = 1024;
  const dm = decimals < 0 ? 0 : decimals;
  const sizes = ['B', 'KB', 'MB', 'GB', 'TB'];
  const i = Math.floor(Math.log(bytes) / Math.log(k));
  return parseFloat((bytes / Math.pow(k, i)).toFixed(dm)) + ' ' + sizes[i];
}

export function formatDate(dateString) {
  if (!dateString) return '-';
  const date = new Date(dateString);
  const y = date.getFullYear();
  const m = String(date.getMonth() + 1).padStart(2, '0');
  const d = String(date.getDate()).padStart(2, '0');
  const hh = String(date.getHours()).padStart(2, '0');
  const mm = String(date.getMinutes()).padStart(2, '0');
  return `${y}-${m}-${d} ${hh}:${mm}`;
}
```

#### 2. `src/utils/md5.js`
使用 `Spark-MD5` 按分块异步读取并计算文件 MD5，避免浏览器因大文件内存溢出。

```javascript
import SparkMD5 from 'spark-md5';

export function calculateMD5(file, onProgress) {
  return new Promise((resolve, reject) => {
    const blobSlice = File.prototype.slice || File.prototype.mozSlice || File.prototype.webkitSlice;
    const chunkSize = 2097152; // 2MB 块大小
    const chunks = Math.ceil(file.size / chunkSize);
    let currentChunk = 0;
    const spark = new SparkMD5.ArrayBuffer();
    const fileReader = new FileReader();

    fileReader.onload = function (e) {
      spark.append(e.target.result);
      currentChunk++;

      if (currentChunk < chunks) {
        if (typeof onProgress === 'function') {
          onProgress(Math.round((currentChunk / chunks) * 100));
        }
        loadNext();
      } else {
        if (typeof onProgress === 'function') {
          onProgress(100);
        }
        resolve(spark.end());
      }
    };

    fileReader.onerror = function () {
      reject('MD5 计算失败，文件读取错误。');
    };

    function loadNext() {
      const start = currentChunk * chunkSize;
      const end = ((start + chunkSize) >= file.size) ? file.size : start + chunkSize;
      fileReader.readAsArrayBuffer(blobSlice.call(file, start, end));
    }

    loadNext();
  });
}
```

#### 3. `src/api/index.js`
带拦截器的统一 Axios 实例。

```javascript
import axios from 'axios';
import { ElMessage } from 'element-plus';
import router from '../router';
import { useUserStore } from '../stores/user';

const request = axios.create({
  baseURL: 'http://localhost:8088/api',
  timeout: 15000
});

// 请求拦截器
request.interceptors.request.use(
  (config) => {
    const userStore = useUserStore();
    if (userStore.token) {
      config.headers['Authorization'] = `Bearer ${userStore.token}`;
    }
    return config;
  },
  (error) => {
    return Promise.reject(error);
  }
);

// 响应拦截器
request.interceptors.response.use(
  (response) => {
    const res = response.data;
    // 如果返回的不是标准 CommonResult（如 MinIO 直传响应或第三方服务），直接放过
    if (res.code === undefined) {
      return response;
    }
    
    if (res.code === 200) {
      return res;
    }

    // 针对特定状态处理
    ElMessage.error(res.message || '系统错误');
    return Promise.reject(new Error(res.message || 'Error'));
  },
  (error) => {
    if (error.response && error.response.status === 401) {
      const userStore = useUserStore();
      userStore.logout();
      ElMessage.error('登录已过期，请重新登录');
      router.push('/login');
    } else {
      ElMessage.error(error.message || '连接服务器失败');
    }
    return Promise.reject(error);
  }
);

export default request;
```

### 第四部分：Pinia 状态管理 (Stores)

#### 1. `src/stores/user.js`
```javascript
import { defineStore } from 'pinia';

export const useUserStore = defineStore('user', {
  state: () => ({
    token: localStorage.getItem('token') || '',
    userId: localStorage.getItem('userId') || '',
    phone: localStorage.getItem('phone') || '',
    firstLogin: localStorage.getItem('firstLogin') === 'true',
    nickname: localStorage.getItem('nickname') || '',
    avatarUrl: localStorage.getItem('avatarUrl') || ''
  }),
  actions: {
    saveLoginState(data) {
      this.token = data.token;
      this.userId = data.userId;
      this.phone = data.phone;
      this.firstLogin = data.firstLogin;
      
      localStorage.setItem('token', data.token);
      localStorage.setItem('userId', data.userId);
      localStorage.setItem('phone', data.phone);
      localStorage.setItem('firstLogin', String(data.firstLogin));
    },
    updateTokenAfterSetPassword(token) {
      this.token = token;
      this.firstLogin = false;
      localStorage.setItem('token', token);
      localStorage.setItem('firstLogin', 'false');
    },
    setNicknameAndAvatar(nickname, avatarUrl) {
      this.nickname = nickname;
      this.avatarUrl = avatarUrl;
      localStorage.setItem('nickname', nickname);
      localStorage.setItem('avatarUrl', avatarUrl || '');
    },
    logout() {
      this.token = '';
      this.userId = '';
      this.phone = '';
      this.firstLogin = false;
      this.nickname = '';
      this.avatarUrl = '';
      localStorage.clear();
    }
  }
});
```

#### 2. `src/components/file/FileUploadDialog.vue`

```vue
<template>
  <el-dialog
    v-model="visible"
    title="上传家庭文件"
    width="500px"
    :close-on-click-modal="false"
  >
    <div class="dialog-body">
      <el-upload
        class="ap-uploader"
        drag
        action=""
        multiple
        :auto-upload="false"
        :on-change="handleFileChange"
      >
        <el-icon class="el-icon--upload"><UploadFilled /></el-icon>
        <div class="el-upload__text">
          拖拽文件到这里，或<em>点击上传</em>
        </div>
      </el-upload>

      <!-- 任务列表 -->
      <div v-if="tasks.length > 0" class="task-list">
        <div v-for="task in tasks" :key="task.id" class="task-item">
          <div class="task-info">
            <span class="task-name" :title="task.name">{{ task.name }}</span>
            <span class="task-status" :class="task.status">{{ task.statusText }}</span>
          </div>
          <el-progress
            :percentage="task.progress"
            :status="task.progressStatus"
            :stroke-width="4"
          />
        </div>
      </div>
    </div>
  </el-dialog>
</template>

<script setup>
import { ref } from 'vue';
import axios from 'axios';
import { useFileStore } from '../../stores/file';
import { calculateMD5 } from '../../utils/md5';
import { preUpload, confirmUpload } from '../../api/file';
import { ElMessage } from 'element-plus';

const visible = ref(false);
const fileStore = useFileStore();
const tasks = ref([]);

const open = () => {
  visible.value = true;
  tasks.value = [];
};

defineExpose({ open });
const emit = defineEmits(['uploaded']);

const handleFileChange = async (uploadFile) => {
  const file = uploadFile.raw;
  const taskId = Date.now() + Math.random().toString(36).substr(2, 5);
  
  const newTask = {
    id: taskId,
    name: file.name,
    progress: 0,
    status: 'pending',
    statusText: '正在准备计算特征码...',
    progressStatus: ''
  };
  
  tasks.value.push(newTask);
  const task = tasks.value.find(t => t.id === taskId);

  try {
    // 1. MD5 计算
    task.statusText = '计算特征码中...';
    const md5 = await calculateMD5(file, (p) => {
      task.progress = Math.round(p * 0.3); // 特征码阶段占30%进度
    });

    // 2. 秒传判定
    task.statusText = '极速匹配中...';
    const preRes = await preUpload({
      md5,
      fileName: file.name,
      parentId: fileStore.currentFolderId
    });

    if (preRes.data.instant) {
      task.progress = 100;
      task.status = 'success';
      task.statusText = '秒传成功！';
      task.progressStatus = 'success';
      ElMessage.success(`${file.name} 秒传成功`);
      emit('uploaded');
      return;
    }

    // 3. 浏览器直传 MinIO 桶（PUT 方法）
    task.statusText = '开始上传到家庭服务器...';
    const preSignedUrl = preRes.data.preSignedUrl;
    const storageKey = preRes.data.storageKey;

    await axios.put(preSignedUrl, file, {
      headers: { 'Content-Type': file.type || 'application/octet-stream' },
      onUploadProgress: (progressEvent) => {
        const uploadProgress = Math.round((progressEvent.loaded * 100) / progressEvent.total);
        task.progress = 30 + Math.round(uploadProgress * 0.6); // 直传阶段占60%
        task.statusText = `正在极速直传...(${uploadProgress}%)`;
      }
    });

    // 4. 确认写入数据库
    task.statusText = '家庭云存储确认中...';
    await confirmUpload({
      storageKey,
      fileName: file.name,
      parentId: fileStore.currentFolderId,
      md5,
      fileSize: file.size,
      contentType: file.type || 'application/octet-stream'
    });

    task.progress = 100;
    task.status = 'success';
    task.statusText = '上传成功！';
    task.progressStatus = 'success';
    ElMessage.success(`${file.name} 上传成功`);
    emit('uploaded');

  } catch (error) {
    console.error(error);
    task.status = 'exception';
    task.statusText = '操作终止/错误';
    task.progressStatus = 'exception';
    ElMessage.error(`${file.name} 传输失败`);
  }
};
</script>
```

```vue
<template>
  <div class="file-browser">
    <FileToolbar
      :selected-count="selectedFiles.length"
      @upload="triggerUpload"
      @create-folder="triggerCreateFolder"
      @batch-delete="triggerBatchDelete"

            :files="files"

            @item-contextmenu="openContextMenu"
            @item-open="handleItemOpen"

          <template #action>
            <el-button type="primary" :icon="Plus" @click="triggerUpload">上传文件</el-button>
          </template>
        </EmptyState>
      </template>
    </el-skeleton>

    <!-- 极速右键功能浮动菜单 -->
    <FileContextMenu
      v-model:visible="contextMenuVisible"
      :x="contextMenuX"
      :y="contextMenuY"
      :active-file="rightClickedFile"
      @action="handleMenuAction"
    />

    <!-- 统一业务对话框 -->
    <FileUploadDialog ref="uploadDialogRef" @uploaded="loadData" />
    <FolderCreateDialog ref="folderDialogRef" @created="loadData" />
    <FilePreviewDialog ref="previewDialogRef" />

    <!-- 极速共享链接展示弹窗 -->
    <el-dialog v-model="shareVisible" title="共享创建成功 🍂" width="450px">
      <div class="share-box">
        <p>全家成员及访客均可通过此链接极速直取：</p>
        <el-input :model-value="shareLink" readonly class="share-input">
          <template #append>
            <el-button @click="copyShareLink">复制</el-button>
          </template>
        </el-input>
        <p class="expire-tip">有效截止日期：{{ formatDate(shareExpireTime) }}</p>
      </div>
    </el-dialog>
  </div>
</template>

<script setup>
import { ref, watch } from 'vue';
import { Plus } from '@element-plus/icons-vue';
import { useFileStore } from '../stores/file';
import { getFileList, deleteFile, batchDeleteFiles, downloadFile } from '../api/file';
import { addFavorite } from '../api/favorite';
import { createShareLink } from '../api/share';
import { formatDate } from '../utils/format';
import { ElMessage, ElMessageBox } from 'element-plus';

import FileToolbar from '../components/file/FileToolbar.vue';
import FileGridView from '../components/file/FileGridView.vue';
import FileListView from '../components/file/FileListView.vue';
import EmptyState from '../components/common/EmptyState.vue';
import FileContextMenu from '../components/file/FileContextMenu.vue';
import FileUploadDialog from '../components/file/FileUploadDialog.vue';
import FolderCreateDialog from '../components/file/FolderCreateDialog.vue';
import FilePreviewDialog from '../components/file/FilePreviewDialog.vue';

const fileStore = useFileStore();

const loading = ref(false);
const files = ref([]);
const selectedFiles = ref([]);

// 右键菜单状态
const contextMenuVisible = ref(false);
const contextMenuX = ref(0);
const contextMenuY = ref(0);
const rightClickedFile = ref(null);

// 对话框引用
const uploadDialogRef = ref(null);
const folderDialogRef = ref(null);
const previewDialogRef = ref(null);

// 共享
const shareVisible = ref(false);
const shareLink = ref('');
const shareExpireTime = ref('');

const loadData = async () => {
  loading.value = true;
  selectedFiles.value = [];
  try {
    const res = await getFileList(fileStore.currentFolderId);
    files.value = res.data;
  } catch (error) {
    console.error(error);
  } finally {
    loading.value = false;
  }
};

watch(() => fileStore.currentFolderId, loadData, { immediate: true });

// 上传与建文件夹
const triggerUpload = () => uploadDialogRef.value?.open();
const triggerCreateFolder = () => folderDialogRef.value?.open();

// 批量删除
const triggerBatchDelete = () => {
  if (selectedFiles.value.length === 0) return;
  ElMessageBox.confirm(
    `确定要把这 ${selectedFiles.value.length} 个文件移入垃圾站吗？`,
    '移入垃圾站',
    { confirmButtonText: '确定', cancelButtonText: '取消', type: 'warning' }
  ).then(async () => {
    try {
      const ids = selectedFiles.value.map(f => f.id);
      await batchDeleteFiles(ids);
      ElMessage.success('已全部安全移入垃圾站');
      loadData();
    } catch (e) {
      console.error(e);
    }
  });
};

// 各种交互映射
const toggleItemSelection = (file) => {
  const index = selectedFiles.value.findIndex(f => f.id === file.id);
  if (index > -1) {
    selectedFiles.value.splice(index, 1);
  } else {
    selectedFiles.value.push(file);
  }
};

const handleListSelection = (selection) => {
  selectedFiles.value = selection;
};

const handleItemOpen = (file) => {
  if (file.isFolder === 1) {
    fileStore.setCurrentFolder(file.id);
  } else {
    previewDialogRef.value?.open(file);
  }
};

// 右键
const openContextMenu = (event, file) => {
  rightClickedFile.value = file;
  contextMenuX.value = event.clientX;
  contextMenuY.value = event.clientY;
  contextMenuVisible.value = true;
};

// 菜单选项路由动作
const handleMenuAction = async ({ action, file }) => {
  if (action === 'open') {
    handleItemOpen(file);
  } else if (action === 'download') {
    try {
      const res = await downloadFile(file.id);
      window.open(res.data.downloadUrl);
    } catch (err) {
      console.error(err);
    }
  } else if (action === 'favorite') {
    try {
      await addFavorite(file.id);
      ElMessage.success('已加入全家收藏');
    } catch (err) {
      console.error(err);
    }
  } else if (action === 'share') {
    try {
      const res = await createShareLink(file.id, 24);
      shareLink.value = `${window.location.origin}/share/${res.data.shareCode}`;
      shareExpireTime.value = res.data.expireTime;
      shareVisible.value = true;
    } catch (err) {
      console.error(err);
    }
  } else if (action === 'delete') {
    ElMessageBox.confirm(`确定要将“${file.fileName}”移至垃圾站吗？`, '警告', {
      type: 'warning'
    }).then(async () => {
      await deleteFile(file.id);
      ElMessage.success('已安全放入垃圾箱');
      loadData();
    });
  } else if (action === 'rename') {
    ElMessageBox.prompt('请输入新的名称', '重命名', {
      inputValue: file.fileName,
      confirmButtonText: '确定',
      cancelButtonText: '取消'
    }).then(async ({ value }) => {
      if (value.trim()) {
        const { renameFile } = await import('../api/file');
        await renameFile(file.id, value.trim());
        ElMessage.success('名称修改成功');
        loadData();
      }
    });
  } else if (action === 'move') {
    ElMessage.info('本系统支持移动至目标目录，前端可在后续补充完整的目录树选择交互。');
  }
};

const copyShareLink = () => {
  navigator.clipboard.writeText(shareLink.value);
  ElMessage.success('直链已极速复制到剪切板，去发给家人吧！');
};
</script>

}

  margin: 16px 0;
}
.expire-tip {

}
</style>
```
