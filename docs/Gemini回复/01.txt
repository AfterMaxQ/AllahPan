### 3. 网络与拦截器

**`src/api/index.js`**
统一处理 JWT 携带，以及自动解析后端 `CommonResult` 包装结构。

```javascript
import axios from 'axios'
import { useUserStore } from '@/stores/user'
import { ElMessage } from 'element-plus'
import router from '@/router'

const request = axios.create({
  baseURL: '/api',
  timeout: 30000
})

// 请求拦截：携带 Token
request.interceptors.request.use(config => {
  const userStore = useUserStore()
  if (userStore.token) {
    config.headers.Authorization = `Bearer ${userStore.token}`
  }
  return config
})

// 响应拦截：处理 CommonResult 和 401
request.interceptors.response.use(
  response => {
    const res = response.data
    // 兼容不需要 CommonResult 包装的 search-service
    if (res.code === undefined && res.success !== undefined) return res

    if (res.code === 200) {
      return res.data
    } else {
      ElMessage.error(res.message || '系统错误')
      return Promise.reject(new Error(res.message || 'Error'))
    }
  },
  error => {
    if (error.response?.status === 401) {
      const userStore = useUserStore()
      userStore.logout()
      router.push('/login')
      ElMessage.warning('登录已过期，请重新登录')
    } else {
      ElMessage.error(error.message || '网络请求失败')
    }
    return Promise.reject(error)
  }
)

export default request
```

### 4. 状态管理 (Pinia)

**`src/stores/user.js`**
管理用户会话、登录态与持久化。

```javascript
import { defineStore } from 'pinia'
import { ref } from 'vue'

export const useUserStore = defineStore('user', () => {
  const token = ref(localStorage.getItem('allahpan_token') || '')
  const userInfo = ref(null)
  const isFirstLogin = ref(localStorage.getItem('allahpan_first') === 'true')

  const setAuth = (newToken, firstLogin) => {
    token.value = newToken
    isFirstLogin.value = firstLogin
    localStorage.setItem('allahpan_token', newToken)
    localStorage.setItem('allahpan_first', firstLogin)
  }

  const logout = () => {
    token.value = ''
    userInfo.value = null
    isFirstLogin.value = false
    localStorage.removeItem('allahpan_token')
    localStorage.removeItem('allahpan_first')
  }

  return { token, userInfo, isFirstLogin, setAuth, logout }
})
```

```javascript
import { useUserStore } from '@/stores/user'

const routes = [
  {
    path: '/login',

    component: () => import('@/views/Login.vue'),

  },

      { path: 'favorites', name: 'Favorites', component: () => import('@/views/Favorites.vue') },
      { path: 'search', name: 'Search', component: () => import('@/views/Search.vue') }
    ]
  }
]

const router = createRouter({
  history: createWebHistory(),
  routes
})

router.beforeEach((to, from, next) => {
  const userStore = useUserStore()
  
  if (!to.meta.public && !userStore.token) {
    return next('/login')
  }
  
  // 首次登录必须设置密码
  if (userStore.token && userStore.isFirstLogin && to.path !== '/set-password') {
    return next('/set-password')
  }

  // 已登录不能再去登录页
  if (to.path === '/login' && userStore.token) {
    return next('/')
  }

  next()
})

export default router
```

### 6. MD5 与文件直传核心逻辑

这是云盘最关键的技术点：前端计算 MD5，请求预签名，直传 MinIO，最后确认。

**`src/utils/md5.js`**
```javascript
import SparkMD5 from 'spark-md5'

// 计算文件大文件 MD5 (分片读取)
export const calculateFileMD5 = (file) => {
  return new Promise((resolve, reject) => {
    const chunkSize = 2 * 1024 * 1024 // 2MB
    const chunks = Math.ceil(file.size / chunkSize)
    let currentChunk = 0
    const spark = new SparkMD5.ArrayBuffer()
    const fileReader = new FileReader()

    fileReader.onload = (e) => {
      spark.append(e.target.result)
      currentChunk++
      if (currentChunk < chunks) {
        loadNext()
      } else {
        resolve(spark.end())
      }
    }
    fileReader.onerror = () => reject(new Error('文件读取失败'))

    const loadNext = () => {
      const start = currentChunk * chunkSize
      const end = Math.min(start + chunkSize, file.size)
      fileReader.readAsArrayBuffer(file.slice(start, end))
    }
    loadNext()
  })
}
```

**`src/api/file.js`**
```javascript
import request from './index'

export const fileApi = {
  // 预上传检测
  preUpload: (data) => request.post('/file/pre-upload', data),
  
  // 确认上传
  confirmUpload: (data) => request.post('/file/confirm-upload', data),
  
  // MinIO 直传 (不经过拦截器，使用原生 fetch)
  uploadToMinio: async (url, file) => {
    const res = await fetch(url, {
      method: 'PUT',
      body: file,
      headers: {
        // 重要：不传特殊 header，如果有 content-type 需要和预签名一致
        'Content-Type': file.type || 'application/octet-stream'
      }
    })
    if (!res.ok) throw new Error('MinIO 上传失败')
  },

  // 获取文件列表
  getList: (parentId = 0) => request.get(`/file/list?parentId=${parentId}`),
  
  // 创建文件夹
  createFolder: (folderName, parentId) => request.post('/file/create-folder', { folderName, parentId })
}
```

### 实施指南

这套代码严格遵循了你的要求。你接下来的步骤是：

1. `npm create vite@latest allahpan-web -- --template vue` 
2. `npm install axios vue-router pinia element-plus @element-plus/icons-vue spark-md5 sass-embedded`
3. 将上述内容分别创建到对应目录即可。

**关于代码设计的注意事项：**
- **MD5与上传**：我在 `FileBrowser.vue` 中实现了 `handleFileUpload` 方法。这真实反映了后端 `pre-upload -> minio PUT -> confirm-upload` 的全链路设计。`fetch` 用于直接请求 MinIO，因为 axios 全局注入了 JWT，如果发给 MinIO 会导致 AWS v4 Signature 失效。
- **UI 设计**：`global.css` 全局注入了原木色和暖白底色，且覆盖了 Element Plus 的核心变量。`.allahpan-card` 提供了一致的柔和阴影和悬浮弹性交互。
- **首次登录拦截**：在 `router/index.js` 中做了严格控制，若 `isFirstLogin` 为 true，必须重定向到 `/set-password` 且不能跳出。

### 2. 基础工具类

**`src/utils/format.js`**
处理文件大小和时间的格式化，符合设计规范的要求。
```javascript
export const formatFileSize = (bytes) => {
  if (bytes === 0) return '0 B'
  if (bytes < 1024) return bytes + ' B'
  if (bytes < 1024 * 1024) return (bytes / 1024).toFixed(1) + ' KB'
  if (bytes < 1024 * 1024 * 1024) return (bytes / (1024 * 1024)).toFixed(1) + ' MB'
  return (bytes / (1024 * 1024 * 1024)).toFixed(2) + ' GB'
}

export const formatDate = (isoString) => {
  if (!isoString) return ''
  const date = new Date(isoString)
  return new Intl.DateTimeFormat('zh-CN', {
    year: 'numeric', month: '2-digit', day: '2-digit',
    hour: '2-digit', minute: '2-digit'
  }).format(date)
}
```

### 4. 首次登录设置密码页

**`src/views/SetPassword.vue`**
新用户首次用验证码登录后，会被路由守卫强制拦截到此页。
```vue
<template>
  <div class="login-container">
    <div class="login-box allahpan-card">
      <div class="header">
        <el-icon size="40" color="var(--el-color-primary)"><Key /></el-icon>
        <h2>设置登录密码</h2>
        <p class="subtitle">欢迎加入 AllahPan，请为您的账号设置一个密码</p>
      </div>

      <el-form :model="form" @submit.prevent="handleSubmit">
        <el-form-item>
          <el-input v-model="form.password" type="password" placeholder="请输入新密码" size="large" show-password>
            <template #prefix><el-icon><Lock /></el-icon></template>
          </el-input>
        </el-form-item>
        <el-form-item>
          <el-input v-model="form.confirmPassword" type="password" placeholder="请再次确认密码" size="large" show-password>
            <template #prefix><el-icon><CircleCheck /></el-icon></template>
          </el-input>
        </el-form-item>
        <el-button type="primary" size="large" class="submit-btn" @click="handleSubmit">开启云盘之旅</el-button>
      </el-form>
    </div>
  </div>
</template>

<script setup>
import { ref } from 'vue'
import { useRouter } from 'vue-router'
import { useUserStore } from '@/stores/user'
import { userApi } from '@/api/user'
import { ElMessage } from 'element-plus'

const router = useRouter()
const userStore = useUserStore()
const form = ref({ password: '', confirmPassword: '' })

const handleSubmit = async () => {
  if (form.value.password.length < 6) return ElMessage.warning('密码至少6位')
  if (form.value.password !== form.value.confirmPassword) return ElMessage.warning('两次输入密码不一致')

  try {
    const res = await userApi.setPassword(form.value.password)
    // 更新 token，并将 firstLogin 标志置为 false
    userStore.setAuth(res.token, false)
    ElMessage.success('密码设置成功！')
    router.replace('/')
  } catch (error) {}
}
</script>
```
